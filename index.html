// Lymar: Zero-Cost Error Unions & Concurrency
import std.game as game;
import std.net as net;

fn fetch_data(url: str): str? {
    if (len(url) == 0) {
        return err();
    }
    return ok("HTTP 200 OK: Data Payload");
}

fn main(): int {
    var res: str = fetch_data("https://lymar.org")? else {
        print("Failed to fetch resource");
        return 1;
    };

    print("Fetched: {res}");
    return 0;
}

Designed For Excellence

Why Software Engineers Choose Lymar

Built from the ground up to solve memory management complexity, runtime error bloat, and thread synchronization hazards.

Zero-Cost Error Unions (Type?)

Native error handling combining optionals and fallible results without generic wrapper bloat. Use ok(val), err(), ? propagation, and inline ? else fallbacks.

Deterministic Region Memory

Scope-bound automatic memory management with zero tracing garbage collection overhead. Compiler-inferred lifetimes guarantee predictable performance.

Structured Concurrency

Lexically scoped parallel CPU blocks and concurrent I/O task blocks with typed channel primitives eliminate thread leaks and race conditions.

Pure Composition OOP

No class inheritance overhead. Reusable mixin state blocks, embed promotion, strict : trait conformance, and explicit pub/prot visibility.

Dual Execution Backends

High-performance Register VM for instant execution and debugging (lymar run) plus Fyra AOT native compilation (lymar build).

Built-in IMGUI & Graphics

First-class Retained & Immediate-mode GUI frameworks (std.ui & std.game) with zero external DOM dependencies for game and desktop applications.

// 1. Variables, Constants, and String Interpolation
var name: str = "Alice";
val age: int = 30;

print("User {name} is {age} years old.");

// 2. Dynamic Collections
var fruits: [str] = ["apple", "banana", "orange"];
fruits.append("mango");

iter (item in fruits) {
    print("Fruit: {item}");
}
// Fallible functions return zero-cost Type? error unions
fn divide(a: int, b: int): int? {
    if (b == 0) {
        return err();
    }
    return ok(a / b);
}

fn calculate(): int {
    var safe_val: int = divide(10, 0)? else {
        print("Division failed! Using default fallback.");
        return 0;
    };
    return safe_val;
}
// Lexically scoped structured concurrency
var ch = channel();

concurrent(ch = ch) {
    task(i in 1..3) {
        ch.send("Worker task {i} completed");
    }
}

iter (msg in ch) {
    print("Received: {msg}");
}
// Native Frame and Trait declarations (exported by default without pub)
trait Speaker {
    fn speak();
}

frame Robot: Speaker {
    pub var model: str;

    pub init(m: str) {
        self.model = m;
    }

    pub fn speak() {
        print("Beep boop! Model {self.model} online.");
    }
}

var bot = Robot("RX-78");
bot.speak();

Install Lymar Today

Supports Linux x86-64, macOS (GCC/Clang), and Windows (MSYS2 MinGW64).

Quick One-Line Installation (Linux / macOS / MSYS2):
bash scripts/install.sh --prefix "$HOME/.local" --jobs 4
Build from Source (Git Repository):
$ git clone https://github.com/lymarlang/lymar.git
$ cd lymar && make -j4
$ ./bin/lymar run examples/text_editor.lm
