Real-World Lymar Applications

Explore production-ready software written entirely in Lymar. From GUI text editors and multi-threaded HTTP web servers to Sokol accelerated game UI engines.

1. Desktop GUI Text Editor (examples/text_editor.lm)

Full-featured native GUI text editor featuring multi-line cursor editing, undo/redo buffers, file I/O, and line numbering.

lymar run examples/text_editor.lm 
import std.ui as ui;
import std.io.file as file;

frame TextEditorApp {
    pub var buffer: str;
    pub var file_path: str;
    pub var cursor_line: int;

    pub init(path: str) {
        self.file_path = path;
        self.buffer = file.read_to_string(path)? else {
            return "// New Document\nfn main() {\n    print(\"Hello Lymar!\");\n}";
        };
        self.cursor_line = 1;
    }

    pub fn render() {
        ui.window("Lymar Code Editor - {self.file_path}", 900, 650);
        ui.text_area(self.buffer);
    }
}

fn main(): int {
    var app = TextEditorApp("main.lm");
    app.render();
    return 0;
}

2. Concurrent Multi-Threaded HTTP Web Server (examples/web_server.lm)

High-throughput HTTP server handling channel-based request pools and non-blocking TCP socket responses.

lymar run examples/web_server.lm 
import std.net.tcp as tcp;

fn handle_request(client_id: int): str {
    return "HTTP/1.1 200 OK\r\nContent-Type: text/plain\r\n\r\nHello from Lymar worker #{client_id}";
}

fn main(): int {
    var req_channel = channel();

    print("Starting Lymar HTTP Server on port 8080...");

    concurrent(ch = req_channel, mode = batch, cores = 8) {
        task(req_id in 1..100) {
            val response = handle_request(req_id);
            req_channel.send(response);
        }
    }

    iter (res in req_channel) {
        print("Dispatched: {len(res)} bytes");
    }
    return 0;
}

3. Sokol Hardware Accelerated Game & Telemetry Showcase (tests/game_showcase.lm)

Accelerated GLES/Metal graphics rendering with Truetype typography and instant particle sprite batching.

lymar run tests/game_showcase.lm 
import std.game.app as game;
import std.game.gui as gui;

fn main(): int {
    var engine = game.create_engine("Lymar Sokol Game Engine", 1280, 720);

    while (engine.is_running()) {
        engine.begin_frame();
        gui.text("FPS: 144.0 | Frame Time: 0.006ms");
        engine.end_frame();
    }
    return 0;
}
