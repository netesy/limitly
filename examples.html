Desktop Text Editor (examples/text_editor.lm)

Full-featured GUI text editor featuring multi-line cursor editing, undo/redo buffers, file I/O, and line numbering.

lymar run examples/text_editor.lm 
import std.ui as ui;
import std.text.controller as text_ctrl;

frame TextEditorApp {
    pub var buffer: str;
    pub var cursor_line: int;

    pub init() {
        self.buffer = "// Lymar Text Editor\nfn main() {\n    print(\"Hello!\");\n}";
        self.cursor_line = 1;
    }

    pub fn render() {
        ui.window("Lymar Editor", 800, 600);
        ui.text_area(self.buffer);
    }
}

fn main(): int {
    var app = TextEditorApp();
    app.render();
    return 0;
}

main();

Retained & IMGUI Widget Suite (tests/gui_showcase.lm)

Showcases buttons, sliders, tab navigation, drop shadows, and responsive layout containers.

lymar build tests/gui_showcase.lm 
import std.ui as ui;

fn main(): int {
    ui.init_window("UI Showcase", 1024, 768);

    var volume: float = 0.75;
    if (ui.slider("Volume", volume, 0.0, 1.0)) {
        print("New volume setting: {volume}");
    }
    return 0;
}

main();
